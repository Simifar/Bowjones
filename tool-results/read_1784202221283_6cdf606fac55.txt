    87→    87→    }
    88→    88→  } catch (e) {
    89→    89→    console.error('Failed to reload settings:', e);

    99→    99→  } catch (e) {
   100→   100→    console.error('Failed to reload schedules:', e);
   101→   101→  }
   102→   102→
   103→   103→  // Check mute expiry
   104→   104→  if (settings.isMuted && settings.muteUntil) {
   105→   105→    const muteEnd = new Date(settings.muteUntil);
   106→   106→    if (new Date() > muteEnd) {
   107→   107→      isMuted = false;
   108→   108→      try {
   109→   109→        db.query("UPDATE BotSettings SET isMuted = 0, muteUntil = NULL WHERE id = 'main'").run();
   110→   110→      } catch {}
   111→   111→    } else {
   112→   112→      isMuted = true;
   113→   113→      muteUntil = muteEnd;
   114→   114→    }
   115→   115→  } else {
   116→   116→    isMuted = settings.isMuted;
   117→   117→    muteUntil = null;
   118→   118→  }

   122→   122→async function telegramApi(method: string, payload: any = {}): Promise<any> {
   123→   123→  if (!settings.telegramBotToken) {
   124→   124→    throw new Error('Telegram bot token not configured');
   125→   125→  }
   126→   126→  const url = `https://api.telegram.org/bot${settings.telegramBotToken}/${method}`;
   127→   127→  const resp = await fetch(url, {
   128→   128→    method: 'POST',
   129→   129→    headers: { 'Content-Type': 'application/json' },
   130→   130→    body: JSON.stringify(payload),
   131→   131→  });

   133→   133→  if (!data.ok) {
   134→   134→    throw new Error(`Telegram API error: ${data.description || JSON.stringify(data)}`);
   135→   135→  }
   136→   136→  return data.result;
   137→   137→}

   164→   164→async function ytimesApi(endpoint: string, method: string = 'GET', body?: any): Promise<any> {
   165→   165→  if (!settings.ytimesApiKey) {
   166→   166→    throw new Error('Ytimes API key not configured');

   169→   169→  const headers: Record<string, string> = {
   170→   170→    'Accept': 'application/json;charset=UTF-8',
   171→   171→    'Authorization': settings.ytimesApiKey,
   172→   172→  };
   173→   173→
   174→   174→  const options: RequestInit = { method, headers };
   175→   175→  if (body && method !== 'GET') {
   176→   176→    headers['Content-Type'] = 'application/json;charset=UTF-8';
   177→   177→    options.body = JSON.stringify(body);
   178→   178→  }
   179→   179→
   180→   180→  const resp = await fetch(url, options);
   181→   181→  const data = await resp.json();
   182→   182→
   183→   183→  if (!data.success) {
   184→   184→    throw new Error(`Ytimes API error: ${data.error || 'Unknown error'}`);
   185→   185→  }
   186→   186→  return data;
   187→   187→}

   209→   209→function formatDateDDMMYYYY(date: Date): string {
   210→   210→  const d = String(date.getDate()).padStart(2, '0');
   211→   211→  const m = String(date.getMonth() + 1).padStart(2, '0');
   212→   212→  const y = date.getFullYear();
   213→   213→  return `${d}.${m}.${y}`;
   214→   214→}

   216→   216→function getTodaySchedule(): ShiftScheduleRow | undefined {

   221→   221→  const ourDay = jsDay === 0 ? 6 : jsDay - 1;
   222→   222→  return schedules.find(s => s.dayOfWeek === ourDay && s.isEnabled);
   223→   223→}

   238→   238→async function checkShifts(): Promise<{ checked: boolean; message: string; notifications: any[] }> {
   239→   239→  if (checkInProgress) {
   240→   240→    return { checked: false, message: 'Check already in progress', notifications: [] };
   241→   241→  }
   242→   242→
   243→   243→  if (!settings.ytimesApiKey || !settings.telegramBotToken) {
   244→   244→    return { checked: false, message: 'Bot not fully configured (missing API key or token)', notifications: [] };
   245→   245→  }
   246→   246→
   247→   247→  checkInProgress = true;
   248→   248→  const notifications: any[] = [];
   249→   249→
   250→   250→  try {
   251→   251→    const todaySchedule = getTodaySchedule();
   252→   252→    if (!todaySchedule) {
   253→   253→      lastCheckTime = new Date();
   254→   254→      return { checked: true, message: 'No schedule for today', notifications: [] };
   255→   255→    }
   256→   256→
   257→   257→    const tzDate = getTimezoneDate();
   258→   258→    const [startHour, startMinute] = todaySchedule.startTime.split(':').map(Number);
   259→   259→    const shiftStart = new Date(tzDate);
   260→   260→    shiftStart.setHours(startHour, startMinute, 0, 0);
   261→   261→
   262→   262→    const now = tzDate;
   263→   263→    const minutesSinceStart = Math.floor((now.getTime() - shiftStart.getTime()) / 60000);
   264→   264→
   265→   265→    // Only check if we're past the notification delay
   266→   266→    if (minutesSinceStart < settings.notificationDelay) {
   267→   267→      lastCheckTime = new Date();
   268→   268→      return {
   269→   269→        checked: true,
   270→   270→        message: `Shift not yet due (starts at ${todaySchedule.startTime}, ${minutesSinceStart}min elapsed, need ${settings.notificationDelay}min)`,
   271→   271→        notifications: [],
   272→   272→      };
   273→   273→    }
   274→   274→
   275→   275→    // Don't check if shift started more than 3 hours ago
   276→   276→    if (minutesSinceStart > 180) {
   277→   277→      lastCheckTime = new Date();
   278→   278→      return { checked: true, message: 'Shift started more than 3 hours ago, skipping', notifications: [] };
   279→   279→    }
   280→   280→
   281→   281→    // Check if we already notified today
   282→   282→    const todayStr = formatDateDDMMYYYY(tzDate);
   283→   283→    const existingCheck = db.query(
   284→   284→      'SELECT * FROM ShiftCheck WHERE checkDate = ? LIMIT 1'
   285→   285→    ).get(todayStr) as any;
   286→   286→
   287→   287→    if (existingCheck && existingCheck.notifiedAt) {
   288→   288→      lastCheckTime = new Date();
   289→   289→      return { checked: true, message: 'Already notified for today', notifications: [] };
   290→   290→    }
   291→   291→
   292→   292→    // Fetch shops from Ytimes
   293→   293→    let shops: YtimesShop[] = [];
   294→   294→    try {
   295→   295→      shops = await getShops();
   296→   296→    } catch (e: any) {
   297→   297→      console.error('[YTIMES] Failed to fetch shops:', e.message);
   298→   298→      return { checked: true, message: `Ytimes API error: ${e.message}`, notifications: [] };
   299→   299→    }
   300→   300→
   301→   301→    // Cache shops
   302→   302→    const upsertShop = db.prepare(`
   303→   303→      INSERT INTO TradePoint (id, guid, name, type, cityName, address, phone, updatedAt)
   304→   304→      VALUES (?, ?, ?, ?, ?, ?, ?, datetime('now'))
   305→   305→      ON CONFLICT(guid) DO UPDATE SET
   306→   306→        name = excluded.name, type = excluded.type, cityName = excluded.cityName,
   307→   307→        address = excluded.address, phone = excluded.phone, updatedAt = datetime('now')
   308→   308→    `);
   309→   309→    for (const shop of shops) {
   310→   310→      try {
   311→   311→        upsertShop.run(generateId(), shop.guid, shop.name, shop.type, shop.cityName, shop.address, shop.phone || '');
   312→   312→      } catch {}
   313→   313→    }
   314→   314→
   315→   315→    // Fetch today's shifts
   316→   316→    let shifts: YtimesShift[] = [];
   317→   317→    try {
   318→   318→      shifts = await getShifts(todayStr, todayStr);
   319→   319→    } catch (e: any) {
   320→   320→      console.error('[YTIMES] Failed to fetch shifts:', e.message);
   321→   321→      return { checked: true, message: `Ytimes shift API error: ${e.message}`, notifications: [] };
   322→   322→    }
   323→   323→
   324→   324→    // Check which shops have open shifts
   325→   325→    const openShiftShopGuids = new Set(shifts.map(s => s.shopGuid));
   326→   326→
   327→   327→    // Find shops that don't have an open shift
   328→   328→    const closedShops = shops.filter(s => !openShiftShopGuids.has(s.guid));
   329→   329→
   330→   330→    if (closedShops.length === 0) {
   331→   331→      db.query(`
   332→   332→        INSERT OR REPLACE INTO ShiftCheck (id, shopGuid, checkDate, notifiedAt, isShiftOpen, createdAt)
   333→   333→        VALUES (?, 'all', ?, datetime('now'), 1, datetime('now'))
   334→   334→      `).run(generateId(), todayStr);
   335→   335→      lastCheckTime = new Date();
   336→   336→      return { checked: true, message: 'All shifts are open', notifications: [] };
   337→   337→    }
   338→   338→
   339→   339→    // There are shops without open shifts - notify!
   340→   340→    const insertNotif = db.prepare(`
   341→   341→      INSERT INTO NotificationLog (id, shopName, shopGuid, shiftDate, scheduledAt, message, sentAt, acknowledged)
   342→   342→      VALUES (?, ?, ?, ?, ?, ?, datetime('now'), 0)
   343→   343→    `);
   344→   344→
   345→   345→    for (const shop of closedShops) {
   346→   346→      const message = `⚠️ <b>Касса не открыта!</b>\n\n` +
   347→   347→        `🏪 <b>${shop.name}</b>\n` +
   348→   348→        `📅 Дата: ${todayStr}\n` +
   349→   349→        `⏰ Смена должна начаться: ${todaySchedule.startTime}\n` +
   350→   350→        `⏱ Прошло: ${minutesSinceStart} мин.\n\n` +
   351→   351→        `<i>Откройте кассу как можно скорее!</i>`;
   352→   352→
   353→   353→      if (!isMuted) {
   354→   354→        for (const chatId of settings.adminChatIds) {
   355→   355→          try {
   356→   356→            await sendMessage(chatId, message);
   357→   357→            notifications.push({ shopName: shop.name, sent: true });
   358→   358→          } catch (e: any) {
   359→   359→            console.error(`Failed to notify for ${shop.name}:`, e.message);
   360→   360→            notifications.push({ shopName: shop.name, sent: false, error: e.message });
   361→   361→          }
   362→   362→        }
   363→   363→      }
   364→   364→
   365→   365→      // Log notification
   366→   366→      try {
   367→   367→        insertNotif.run(generateId(), shop.name, shop.guid, todayStr, todaySchedule.startTime, message);
   368→   368→      } catch (e: any) {
   369→   369→        console.error('Failed to log notification:', e.message);
   370→   370→      }
   371→   371→
   372→   372→      // Also notify via webhook to Next.js
   373→   373→      try {
   374→   374→        await fetch(`http://localhost:3000/api/bot-webhook`, {
   375→   375→          method: 'POST',
   376→   376→          headers: { 'Content-Type': 'application/json' },
   377→   377→          body: JSON.stringify({
   378→   378→            type: 'notification_sent',
   379→   379→            data: {
   380→   380→              shopName: shop.name,
   381→   381→              shopGuid: shop.guid,
   382→   382→              shiftDate: todayStr,
   383→   383→              scheduledAt: todaySchedule.startTime,
   384→   384→              message,
   385→   385→            },
   386→   386→          }),
   387→   387→        });
   388→   388→      } catch {}
   389→   389→    }
   390→   390→
   391→   391→    // Record that we checked
   392→   392→    db.query(`
   393→   393→      INSERT OR REPLACE INTO ShiftCheck (id, shopGuid, checkDate, notifiedAt, isShiftOpen, createdAt)
   394→   394→      VALUES (?, 'all', ?, datetime('now'), 0, datetime('now'))
   395→   395→    `).run(generateId(), todayStr);
   396→   396→
   397→   397→    lastCheckTime = new Date();
   398→   398→    const statusMsg = closedShops.length === shops.length
   399→   399→      ? 'No shifts are open!'
   400→   400→      : `${closedShops.map(s => s.name).join(', ')} — no shift open`;
   401→   401→
   402→   402→    return { checked: true, message: statusMsg, notifications };
   403→   403→  } finally {
   404→   404→    checkInProgress = false;
   405→   405→  }
   406→   406→}

   615→   615→// ============ Scheduled Checker ============
   616→   616→async function scheduledCheck() {
   617→   617→  const todaySchedule = getTodaySchedule();
   618→   618→  if (!todaySchedule) return;
   619→   619→  if (!settings.ytimesApiKey || !settings.telegramBotToken) return;
   620→   620→
   621→   621→  const tzDate = getTimezoneDate();
   622→   622→  const [startHour, startMinute] = todaySchedule.startTime.split(':').map(Number);
   623→   623→  const shiftStart = new Date(tzDate);
   624→   624→  shiftStart.setHours(startHour, startMinute, 0, 0);
   625→   625→
   626→   626→  const minutesSinceStart = Math.floor((tzDate.getTime() - shiftStart.getTime()) / 60000);
   627→   627→
   628→   628→  const shouldCheck =
   629→   629→    minutesSinceStart >= settings.notificationDelay &&
   630→   630→    minutesSinceStart <= 180 &&
   631→   631→    (minutesSinceStart === settings.notificationDelay ||
   632→   632→     minutesSinceStart === settings.notificationDelay + 30 ||
   633→   633→     minutesSinceStart === settings.notificationDelay + 60 ||
   634→   634→     minutesSinceStart === settings.notificationDelay + 90 ||
   635→   635→     minutesSinceStart === settings.notificationDelay + 120 ||
   636→   636→     minutesSinceStart === settings.notificationDelay + 150);
   637→   637→
   638→   638→  if (shouldCheck) {
   639→   639→    const todayStr = formatDateDDMMYYYY(tzDate);
   640→   640→    const existingCheck = db.query(
   641→   641→      'SELECT * FROM ShiftCheck WHERE checkDate = ? LIMIT 1'
   642→   642→    ).get(todayStr) as any;
   643→   643→
   644→   644→    if (!existingCheck || !existingCheck.notifiedAt) {
   645→   645→      console.log(`[SCHED] Running scheduled check (${minutesSinceStart}min since shift start)`);
   646→   646→      await checkShifts();
   647→   647→    }
   648→   648→  }
   649→   649→}
